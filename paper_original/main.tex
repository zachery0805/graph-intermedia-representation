% Marks a passage as a candidate for deletion (page-limit trimming pass) --
% renders in red for the authors/advisor to review; does NOT remove or
% comment out the text. Left in place until a human decides to cut it.
% Environment form (not a simple \newcommand{}{}) so it can safely span
% multiple paragraphs / blank lines in the source.
\newenvironment{maybecut}{\color{red}}{}
% Marks a confirmed correction to previously-wrong body text (e.g. the
% Case-1 O(1) claim a reviewer questioned). \wastext{} strikes through the
% old, incorrect wording so it stays visible; \fixtext{} renders the
% corrected replacement in blue immediately after it. Purely additive new
% text (nothing incorrect being replaced) uses \fixtext{} alone, with no
% strikethrough companion.
\newcommand{\wastext}[1]{\sout{#1}}
\newcommand{\fixtext}[1]{\textcolor{blue}{#1}}
% Environment forms of \maybecut/\fixtext for content spanning multiple
% paragraphs (blank lines), where \wastext's \sout cannot be used.
% \begin{maybecut}...\end{maybecut} = red, candidate for deletion.
% \begin{fixed}...\end{fixed} = blue, modified or newly added text.
\newenvironment{fixed}{\color{blue}}{}
